cmake_minimum_required(VERSION 3.21)
project(GoldPricePluginTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_executable(gold_source_tests source_tests.cpp ../GoldPriceSource.cpp)
target_include_directories(gold_source_tests PRIVATE ..)
target_compile_definitions(gold_source_tests PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0601)
target_compile_options(gold_source_tests PRIVATE /utf-8 /EHsc)
target_link_libraries(gold_source_tests PRIVATE wininet)
add_test(NAME quote_parsers COMMAND gold_source_tests)

add_executable(gold_plugin_tests plugin_tests.cpp ../GoldPricePlugin.cpp ../GoldPriceSource.cpp)
target_include_directories(gold_plugin_tests PRIVATE ..)
target_compile_definitions(gold_plugin_tests PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0601 GOLDPRICE_TESTING)
target_compile_options(gold_plugin_tests PRIVATE /utf-8 /EHsc)
target_link_libraries(gold_plugin_tests PRIVATE wininet gdi32 user32)
add_test(NAME independent_requests COMMAND gold_plugin_tests)
set_tests_properties(independent_requests PROPERTIES TIMEOUT 20)

add_library(gold_lifecycle_fixture SHARED ../dllmain.cpp ../GoldPricePlugin.cpp ../GoldPriceSource.cpp lifecycle_transport.cpp)
target_include_directories(gold_lifecycle_fixture PRIVATE ..)
target_compile_definitions(gold_lifecycle_fixture PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0601 GOLDPRICE_TESTING)
target_compile_options(gold_lifecycle_fixture PRIVATE /utf-8 /EHsc)
target_link_libraries(gold_lifecycle_fixture PRIVATE wininet gdi32 user32)
add_executable(gold_lifecycle_tests lifecycle_tests.cpp)
target_include_directories(gold_lifecycle_tests PRIVATE ..)
target_compile_definitions(gold_lifecycle_tests PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0601)
target_compile_options(gold_lifecycle_tests PRIVATE /utf-8 /EHsc)
add_test(NAME unload_during_request COMMAND gold_lifecycle_tests $<TARGET_FILE:gold_lifecycle_fixture>)
set_tests_properties(unload_during_request PROPERTIES TIMEOUT 15)
